import { createClient } from "npm:@supabase/supabase-js@2"; const corsHeaders = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type", "Access-Control-Allow-Methods": "POST, OPTIONS", }; Deno.serve(async (req: Request) => { if (req.method === "OPTIONS") { return new Response("ok", { headers: corsHeaders }); } const json = (data: unknown, status = 200) => new Response(JSON.stringify(data), { status, headers: { ...corsHeaders, "Content-Type": "application/json" }, }); try { if (req.method !== "POST") { return json({ error: "Método não permitido." }, 405); } const supabaseUrl = Deno.env.get("SUPABASE_URL")!; const supabaseKey = Deno.env.get("SUPABASE_ANON_KEY")!; const infinitePayKey = Deno.env.get("INFINITEPAY_API_KEY"); if (!supabaseUrl || !supabaseKey || !infinitePayKey) { return json({ error: "Configuração incompleta no servidor." }, 500); } const { items } = await req.json(); if (!Array.isArray(items) || items.length === 0 || items.length > 30) { return json({ error: "Carrinho inválido." }, 400); } const quantities = new Map(); for (const item of items) { if ( typeof item.product_id !== "string" || !Number.isInteger(item.quantity) || item.quantity < 1 || item.quantity > 10 ) { return json({ error: "Produto ou quantidade inválidos." }, 400); } quantities.set( item.product_id, (quantities.get(item.product_id) ?? 0) + item.quantity, ); } const ids = [...quantities.keys()]; const supabase = createClient(supabaseUrl, supabaseKey); const { data, error } = await supabase .from("catalog_prices") .select("product_id, price") .in("product_id", ids); if (error || !data || data.length !== ids.length) { return json({ error: "Há produtos sem preço cadastrado. Confira o catálogo.", }, 400); } const products = data.map((product) => { const price = Number(product.price); if (!Number.isFinite(price) || price <= 0) { throw new Error("Preço inválido."); } return { quantity: quantities.get(product.product_id)!, price: Math.round(price * 100), description: product.product_id, }; }); const orderNsu = crypto.randomUUID(); const response = await fetch( "https://api.checkout.infinitepay.io/links", { method: "POST", headers: { "X-Api-Key": infinitePayKey, "Content-Type": "application/json", }, body: JSON.stringify({ order_nsu: orderNsu, redirect_url: "https://jeanpxp-droid.github.io/suplementos-m1-catalogo/", items: products, options: { single_payment: true, payment_methods_enabled: ["credit_card", "pix"], }, }), }, ); const result = await response.json(); if (!response.ok || typeof result.url !== "string") { console.error("Erro ao criar checkout:", response.status); return json({ error: "Não foi possível criar o pagamento." }, 502); } return json({ url: result.url, order_nsu: orderNsu }); } catch (error) { console.error("Erro no checkout:", error); return json({ error: "Erro ao processar o carrinho." }, 500); } });
